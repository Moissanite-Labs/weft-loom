    var v = getComputedStyle(document.documentElement).getPropertyValue(name).trim();
    return v || fallback || '';
  } catch (e) { return fallback || ''; }
};


(function() {
  if (window.__lumiCanvasGuard) return;
  window.__lumiCanvasGuard = true;
  var MAX_PX = 4096, MIN_FONT_PX = 12;
  try {
    var CP = HTMLCanvasElement.prototype;
    var BOX = {width: 'clientWidth', height: 'clientHeight'}, D = {};
    Object.keys(BOX).forEach(function(p) {
      D[p] = Object.getOwnPropertyDescriptor(CP, p);
    });
    var tracking = function(el, p) {
      var box = el[BOX[p]];
      return !el.style[p] && box > 0 && box === D[p].get.call(el) &&
          box === el['__lumiLast' + p];
    };
    var MAX = {width: 'maxWidth', height: 'maxHeight'};
    var PAD = {width: ['paddingLeft', 'paddingRight'],
               height: ['paddingTop', 'paddingBottom']};
    var avail = function(el, p) {
      var pe = el.parentElement;
      if (!pe) return 0;
      var cs = getComputedStyle(pe);
      return pe[BOX[p]] - parseFloat(cs[PAD[p][0]]) -
          parseFloat(cs[PAD[p][1]]);
    };
    Object.keys(BOX).forEach(function(p) {
      var d = D[p];
      if (!d || !d.set || !d.get) return;
      Object.defineProperty(CP, p, {
        configurable: true, enumerable: d.enumerable, get: d.get,
        set: function(v) {
          var n = +v, box = this[BOX[p]];
          if (n > MAX_PX) n = MAX_PX;
          if (n > box && tracking(this, p)) {
            var el = this, pins = {};
            Object.keys(BOX).forEach(function(q) {
              if (tracking(el, q)) pins[q] = avail(el, q) ||
                  el['__lumiBox' + q];
            });
            Object.keys(pins).forEach(function(q) {
              el.style[q] = '100%';
              el.style[MAX[q]] = pins[q] + 'px';
            });
            n = box;
          }
          this['__lumiBox' + p] = box;
          this['__lumiLast' + p] = n;
          d.set.call(this, isNaN(n) ? v : n);
        }
      });
    });
  } catch (e) {}
  try {
    var XP = CanvasRenderingContext2D.prototype;
    var px = /(\d*\.?\d+)px/;
    var floorFont = function(ctx) {
      var f = ctx.font, m = px.exec(f);
      if (m && +m[1] >= 4 && +m[1] < MIN_FONT_PX) ctx.font = f.replace(px, MIN_FONT_PX + 'px');
    };
    ['fillText', 'strokeText', 'measureText'].forEach(function(k) {
      var o = XP[k];
      if (typeof o !== 'function') return;
      XP[k] = function() { floorFont(this); return o.apply(this, arguments); };
    });
  } catch (e) {}
})();


(function() {
  function syncBody() {
    var dt = document.documentElement.getAttribute('data-theme');
    if (document.body && (dt === 'dark' || dt === 'light')) {
      document.body.classList.toggle('dark-mode', dt === 'dark');
    }
  }
  function applyTheme(theme) {
    if (theme !== 'dark' && theme !== 'light') return;
    document.documentElement.setAttribute('data-theme', theme);
    syncBody();
  }
  if (typeof MutationObserver === 'function') {
    new MutationObserver(syncBody).observe(document.documentElement, {
      attributes: true, attributeFilter: ['data-theme']
    });
  }
  if (typeof document.addEventListener === 'function') {
    document.addEventListener('DOMContentLoaded', syncBody);
  }
  try {
    var mq = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)');
    if (mq) {
      if (!document.documentElement.hasAttribute('data-theme')) {
        applyTheme(mq.matches ? 'dark' : 'light');
      } else {
        syncBody();
      }
      if (mq.addEventListener) {
        mq.addEventListener('change', function(e) {
          applyTheme(e.matches ? 'dark' : 'light');
        });
      }
    }
  } catch (e) {}
  window.addEventListener('message', function(e) {
    if (e.data && (e.data.type === 'set-theme' || e.data.type === 'APPLY_THEME') && e.data.theme) {
      applyTheme(e.data.theme);
    }
  });
})();


    (function () {
      function getCssVar(name, fallback) {
        return getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;
      }

      function syncSlider(el) {
        const min = parseFloat(el.min);
        const max = parseFloat(el.max);
        const val = parseFloat(el.value);
        el.style.setProperty('--progress', (((val - min) / (max - min)) * 100).toFixed(1) + '%');
      }

      const cv = document.getElementById('morph-cv');
      const ctx = cv.getContext('2d');
      const inDepth = document.getElementById('in-depth');
      const valDepth = document.getElementById('val-depth');
      const kpiComp = document.getElementById('kpi-comp');
      const kpiTrace = document.getElementById('kpi-trace');
      const kpiRet = document.getElementById('kpi-ret');
      const btnReduce = document.getElementById('btn-reduce');
      const btnMutate = document.getElementById('btn-mutate');
      const armBtns = document.querySelectorAll('.lumi-segmented-btn');

      const TAG_NAMES = ['ERA', 'CON', 'DUP', 'VLV', 'FRC', 'SUP'];
      let arm = 'morph';
      let seedIdx = 1;
      let redSteps = 6;
      let animT = 0;
      let nodes = [];
      let edges = [];

      const E11_RETURNS = {
        morph: { 6: '+33.48', 7: '+35.10', 8: '+36.68', 9: '+39.51' },
        scrambled: { 6: '+5.39', 7: '+4.10', 8: '+2.87', 9: '-0.95' },
        direct: { 6: '+17.44', 7: '+10.20', 8: '+4.02', 9: '-0.82' }
      };

      function buildGraph() {
        nodes = [];
        edges = [];
        const nCount = 14 + (redSteps % 7);
        for (let i = 0; i < nCount; i++) {
          const ang = (i / nCount) * Math.PI * 2 - Math.PI / 2;
          const rad = (i % 2 === 0) ? 0.37 : 0.22;
          const tag = i === 0 ? 3 : ((i * 7 + seedIdx * 3 + redSteps) % 6);
          const aux = Math.sin(i * 1.9 + seedIdx * 1.3 + redSteps * 0.4);
          nodes.push({ x: 0.5 + Math.cos(ang) * rad, y: 0.54 + Math.sin(ang) * rad, tag: tag, aux: aux });
        }
        for (let i = 0; i < nCount; i++) {
          edges.push([i, (i + 1) % nCount, false]);
          if (i % 3 === 0) {
            edges.push([i, (i + 5) % nCount, true]);
          }
        }
        updateKPIs();
      }

      function updateKPIs() {
        const d = parseInt(inDepth.value, 10);
        const dim = Math.pow(2, d);
        valDepth.textContent = dim + '²';
        const rawBytes = dim * dim * 4;
        if (arm === 'direct') {
          kpiComp.textContent = (rawBytes / 1024).toFixed(0) + ' KB · 1.0× (raw floats)';
          kpiTrace.textContent = 'No IC graph · O(1/√P) walk';
        } else {
          const ratio = (rawBytes / 492).toFixed(1);
          const tension = Math.min(82, 28 + redSteps * 4);
          kpiComp.textContent = '492 B · ' + ratio + '× (vs ' + dim + '²)';
          kpiTrace.textContent = redSteps + ' steps · tension ' + tension + ' / 85';
        }
        kpiRet.textContent = E11_RETURNS[arm][d] + ' · drift 0.0000';
      }

      function sampleMatrixVal(rNorm, cNorm, d) {
        if (rNorm < 0.25) {
          return Math.sin(rNorm * 37.0 + cNorm * 29.0) * 0.45;
        }
        if (arm === 'direct') {
          const noiseScale = d === 6 ? 0.45 : (d === 7 ? 0.72 : (d === 8 ? 0.90 : 0.98));
          const coh = (1.0 - noiseScale) * Math.cos((rNorm - cNorm) * Math.PI * 2.0);
          const staticNoise = noiseScale * Math.sin((rNorm * 413.7 + cNorm * 389.3) * (d * 3 + seedIdx));
          return coh + staticNoise * 0.85;
        }
        if (arm === 'scrambled') {
          const h = Math.sin((rNorm * 271.1 + cNorm * 619.7) * (seedIdx + 2));
          return Math.abs(h) > 0.92 ? (h > 0 ? 0.85 : -0.85) : 0.04 * h;
        }
        const phase = (nodes[1] ? nodes[1].aux : 0.4) * 1.6;
        const w1 = Math.cos((rNorm * 1.33 - cNorm) * Math.PI * 2.0 - phase);
        const w2 = 0.42 * Math.cos((rNorm * 2.66 - cNorm * 2.0) * Math.PI * 2.0 + redSteps * 0.35);
        const kron = 0.25 * Math.sign(Math.sin(rNorm * Math.PI * 4) * Math.cos(cNorm * Math.PI * 4));
        return 0.62 * w1 + 0.28 * w2 + 0.22 * kron;
      }

      function draw() {
        const dpr = window.devicePixelRatio || 1;
        const rect = cv.getBoundingClientRect();
        const w = Math.max(300, Math.floor(rect.width));
        const h = Math.max(220, Math.floor(rect.height));
        if (cv.width !== w * dpr || cv.height !== h * dpr) {
          cv.width = w * dpr;
          cv.height = h * dpr;
        }
        ctx.save();
        ctx.scale(dpr, dpr);
        ctx.clearRect(0, 0, w, h);

        animT += 0.03;
        const ink = getCssVar('--on-surface', '#111111');
        const muted = getCssVar('--on-surface-variant', '#666666');
        const primary = getCssVar('--primary', '#3186ff');
        const chart3 = getCssVar('--chart-3', '#228a49');
        const chart8 = getCssVar('--chart-8', '#997112');
        const strokeDef = getCssVar('--stroke-default', '#d4d4d8');

        const d = parseInt(inDepth.value, 10);
        const colW = w / 3;

        ctx.font = '12px "Google Sans Code", monospace';
        ctx.fillStyle = muted;
        ctx.fillText('1. BCP GRAPH (492 B)', 10, 16);
        ctx.fillText('2. PLASM MATRIX (' + Math.pow(2, d) + '²)', colW + 10, 16);
        ctx.fillText('3. CILIARY WAVE FIELD', colW * 2 + 10, 16);

        // Panel 1: Combinator Graph
        const gCx = colW * 0.5;
        const gCy = h * 0.54;
        const gScale = Math.min(colW, h - 24);
        if (arm === 'direct') {
          ctx.strokeStyle = strokeDef;
          ctx.setLineDash([4, 4]);
          ctx.beginPath();
          ctx.arc(gCx, gCy, gScale * 0.32, 0, Math.PI * 2);
          ctx.stroke();
          ctx.setLineDash([]);
          ctx.fillStyle = muted;
          ctx.textAlign = 'center';
          ctx.fillText('Bypassed (raw floats)', gCx, gCy + 4);
          ctx.textAlign = 'left';
        } else {
          edges.forEach(function (e) {
            const n1 = nodes[e[0]];
            const n2 = nodes[e[1]];
            const x1 = (n1.x - 0.5) * gScale + gCx;
            const y1 = (n1.y - 0.54) * gScale + gCy;
            const x2 = (n2.x - 0.5) * gScale + gCx;
            const y2 = (n2.y - 0.54) * gScale + gCy;
            ctx.strokeStyle = e[2] ? primary : muted;
            ctx.globalAlpha = e[2] ? 0.75 : 0.35;
            ctx.lineWidth = e[2] ? 1.6 : 1.1;
            if (e[2]) ctx.setLineDash([3, 3]);
            ctx.beginPath();
            ctx.moveTo(x1, y1);
            ctx.lineTo(x2, y2);
            ctx.stroke();
            ctx.setLineDash([]);
          });
          ctx.globalAlpha = 1.0;

          nodes.forEach(function (nd) {
            const nx = (nd.x - 0.5) * gScale + gCx;
            const ny = (nd.y - 0.54) * gScale + gCy;
            ctx.fillStyle = nd.tag === 3 ? chart3 : (nd.tag === 4 ? primary : ink);
            ctx.globalAlpha = 0.88;
            ctx.beginPath();
            ctx.arc(nx, ny, 9.5, 0, Math.PI * 2);
            ctx.fill();
            ctx.globalAlpha = 1.0;
          });
        }

        // Panel 2: Unfolded Matrix Grid (visualized at resolution proportional to depth)
        const cells = d === 6 ? 16 : (d === 7 ? 24 : (d === 8 ? 32 : 44));
        const mPad = 14;
        const mSz = Math.min(colW - mPad * 2, h - 38);
        const mx0 = colW + (colW - mSz) * 0.5;
        const my0 = 26 + (h - 32 - mSz) * 0.5;
        const stepPx = mSz / cells;

        for (let r = 0; r < cells; r++) {
          for (let c = 0; c < cells; c++) {
            const v = sampleMatrixVal(r / cells, c / cells, d);
            const mag = Math.min(1, Math.abs(v));
            ctx.globalAlpha = 0.14 + 0.80 * mag;
            ctx.fillStyle = v >= 0 ? primary : chart8;
            ctx.fillRect(mx0 + c * stepPx, my0 + r * stepPx, Math.max(1, stepPx - 0.6), Math.max(1, stepPx - 0.6));
          }
        }
        ctx.globalAlpha = 1.0;
        ctx.strokeStyle = chart3;
        ctx.lineWidth = 1.6;
        ctx.setLineDash([4, 3]);
        ctx.strokeRect(mx0, my0, mSz, mSz * 0.25);
        ctx.setLineDash([]);

        // Panel 3: Ciliary Membrane Propulsion Ring
        const rCx = colW * 2.5;
        const rCy = h * 0.55;
        const baseR = Math.min(colW, h - 36) * 0.30;
        const nCilia = d === 6 ? 24 : (d === 7 ? 36 : (d === 8 ? 48 : 64));

        ctx.strokeStyle = strokeDef;
        ctx.lineWidth = 1.2;
        ctx.beginPath();
        ctx.arc(rCx, rCy, baseR, 0, Math.PI * 2);
        ctx.stroke();

        for (let k = 0; k < nCilia; k++) {
          const theta = (k / nCilia) * Math.PI * 2;
          let strokeAmp = 0;
          if (arm === 'morph') {
            strokeAmp = Math.cos(theta - animT * 1.8 + seedIdx * 0.5);
          } else if (arm === 'direct') {
            const decay = d === 6 ? 0.75 : (d === 7 ? 0.42 : (d === 8 ? 0.18 : 0.05));
            strokeAmp = decay * Math.cos(theta - animT * 1.8) + (1 - decay) * Math.sin(k * 97.3 + animT * 4.0);
          } else {
            strokeAmp = (k % 7 === 0) ? Math.sin(animT * 2.5 + k) : 0.06 * Math.cos(k * 31.1);
          }
          const bx = rCx + Math.cos(theta) * baseR;
          const by = rCy + Math.sin(theta) * baseR;
          const len = 10 + 11 * strokeAmp;
          const tang = theta + 0.45 * strokeAmp;
          ctx.strokeStyle = arm === 'morph' ? primary : (Math.abs(strokeAmp) > 0.4 ? chart8 : muted);
          ctx.lineWidth = d >= 8 ? 1.4 : 1.8;
          ctx.beginPath();
          ctx.moveTo(bx, by);
          ctx.lineTo(bx + Math.cos(tang) * len, by + Math.sin(tang) * len);
          ctx.stroke();
        }

        if (arm === 'morph') {
          ctx.strokeStyle = chart3;
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.moveTo(rCx - 12, rCy);
          ctx.lineTo(rCx + 14, rCy);
          ctx.lineTo(rCx + 8, rCy - 5);
          ctx.moveTo(rCx + 14, rCy);
          ctx.lineTo(rCx + 8, rCy + 5);
          ctx.stroke();
        }

        ctx.restore();
        requestAnimationFrame(draw);
      }

      inDepth.addEventListener('input', function () {
        syncSlider(inDepth);
        updateKPIs();
      });

      btnReduce.addEventListener('click', function () {
        redSteps = (redSteps % 18) + 2;
        buildGraph();
      });

      btnMutate.addEventListener('click', function () {
        seedIdx += 1;
        buildGraph();
      });

      armBtns.forEach(function (b) {
        b.addEventListener('click', function () {
          armBtns.forEach(function (x) { x.classList.remove('is-active'); });
          b.classList.add('is-active');
          arm = b.getAttribute('data-arm');
          updateKPIs();
        });
      });

      syncSlider(inDepth);
      buildGraph();
      requestAnimationFrame(draw);
    })();
