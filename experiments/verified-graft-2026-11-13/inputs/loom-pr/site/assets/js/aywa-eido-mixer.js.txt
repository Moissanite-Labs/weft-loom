/**
 * aywa-eido-mixer.js — Eido Continuous Barycentric Style Mixer & Issue #45 Telemetry Provenance.
 * Implements circular `atan2` hue averaging on S^1 (`[0, 1)` turns or `[0, 360)` degrees)
 * so blending red at 0.95 and red at 0.05 yields red (0.00), never cyan (0.50).
 * License: Apache-2.0
 */

export const TELEMETRY_PROVENANCE = Object.freeze({
  MEASURED: "measured",
  DERIVED: "derived",
  FIXTURE: "fixture",
  UNAVAILABLE: "unavailable"
});

export function classifyTelemetrySource({ liveExecuted = false, derivedFromExactFormula = false, staticFixture = false } = {}) {
  if (staticFixture) return TELEMETRY_PROVENANCE.FIXTURE;
  if (liveExecuted && derivedFromExactFormula) return TELEMETRY_PROVENANCE.DERIVED;
  if (liveExecuted) return TELEMETRY_PROVENANCE.MEASURED;
  return TELEMETRY_PROVENANCE.UNAVAILABLE;
}

export function projectOntoSimplex(w1, w2, w3) {
  const c1 = Math.max(0.0, Number(w1) || 0.0);
  const c2 = Math.max(0.0, Number(w2) || 0.0);
  const c3 = Math.max(0.0, Number(w3) || 0.0);
  const s = c1 + c2 + c3;
  if (s <= 1e-12) return [1 / 3, 1 / 3, 1 / 3];
  return [c1 / s, c2 / s, c3 / s];
}

export function barycentricFromPoint(px, py, v1, v2, v3) {
  const det = (v2.y - v3.y) * (v1.x - v3.x) + (v3.x - v2.x) * (v1.y - v3.y);
  if (Math.abs(det) < 1e-12) return [1 / 3, 1 / 3, 1 / 3];
  const raw1 = ((v2.y - v3.y) * (px - v3.x) + (v3.x - v2.x) * (py - v3.y)) / det;
  const raw2 = ((v3.y - v1.y) * (px - v3.x) + (v1.x - v3.x) * (py - v3.y)) / det;
  const raw3 = 1.0 - raw1 - raw2;
  return projectOntoSimplex(raw1, raw2, raw3);
}

/**
 * Circular angular mean over unit-turn hues `h_i in [0, 1)` weighted by normalized weights.
 * Example from w3ft.org: `circularHueMeanTurns([0.95, 0.05], [0.5, 0.5]) === 0.0` (red, NOT 0.5 cyan).
 */
export function circularHueMeanTurns(huesTurns, weights) {
  let sinSum = 0.0;
  let cosSum = 0.0;
  for (let i = 0; i < huesTurns.length; i++) {
    const w = weights[i] ?? 0.0;
    const theta = 2.0 * Math.PI * huesTurns[i];
    sinSum += w * Math.sin(theta);
    cosSum += w * Math.cos(theta);
  }
  let angle = Math.atan2(sinSum, cosSum) / (2.0 * Math.PI);
  if (angle < 0.0) angle += 1.0;
  if (Math.abs(angle - 1.0) < 1e-12) angle = 0.0;
  return angle;
}

export const DEFAULT_EIDO_PRESETS = Object.freeze([
  { name: "radiant", hue: 0.60, rounding: 0.62, density: 0.50, contrast: 0.92, dark: 1.0 },
  { name: "pastel",  hue: 0.08, rounding: 0.88, density: 0.65, contrast: 0.14, dark: 0.0 },
  { name: "terminal", hue: 0.44, rounding: 0.14, density: 0.34, contrast: 0.96, dark: 0.95 }
]);

export function mixAestheticPresets(weights, presets = DEFAULT_EIDO_PRESETS) {
  const normW = projectOntoSimplex(weights[0], weights[1], weights[2]);
  const hue = circularHueMeanTurns(
    presets.map(p => p.hue),
    normW
  );
  let rounding = 0.0, density = 0.0, contrast = 0.0, dark = 0.0;
  for (let i = 0; i < 3; i++) {
    rounding += normW[i] * presets[i].rounding;
    density += normW[i] * presets[i].density;
    contrast += normW[i] * presets[i].contrast;
    dark += normW[i] * presets[i].dark;
  }
  return {
    weights: normW,
    hue,
    hueDeg: hue * 360.0,
    rounding,
    radiusPx: 2.0 + rounding * 20.0,
    density,
    padPx: 9.0 + density * 10.0,
    contrast,
    dark
  };
}

export function mountSimplexPad(svgEl, vertices, onChange) {
  if (!svgEl) return () => {};
  let dragging = false;
  const handle = (evt) => {
    const rect = svgEl.getBoundingClientRect();
    if (rect.width <= 0 || rect.height <= 0) return;
    const vb = svgEl.viewBox.baseVal;
    const px = ((evt.clientX - rect.left) / rect.width) * (vb.width || rect.width);
    const py = ((evt.clientY - rect.top) / rect.height) * (vb.height || rect.height);
    const w = barycentricFromPoint(px, py, vertices[0], vertices[1], vertices[2]);
    onChange(mixAestheticPresets(w), w);
  };
  const down = (e) => { dragging = true; handle(e); };
  const move = (e) => { if (dragging) handle(e); };
  const up = () => { dragging = false; };
  svgEl.addEventListener("pointerdown", down);
  svgEl.addEventListener("pointermove", move);
  window.addEventListener("pointerup", up);
  return () => {
    svgEl.removeEventListener("pointerdown", down);
    svgEl.removeEventListener("pointermove", move);
    window.removeEventListener("pointerup", up);
  };
}
