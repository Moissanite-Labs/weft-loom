export function sha256Bytes(inputBytes) {
  const bytes = inputBytes instanceof Uint8Array
    ? inputBytes
    : new TextEncoder().encode(String(inputBytes));
  const bitLen = bytes.length * 8;
  const padLen = (bytes.length + 9 + 63) & ~63;
  const buf = new Uint8Array(padLen);
  buf.set(bytes);
  buf[bytes.length] = 0x80;
  const dv = new DataView(buf.buffer);
  dv.setUint32(padLen - 8, Math.floor(bitLen / 0x100000000), false);
  dv.setUint32(padLen - 4, bitLen >>> 0, false);

  let h0 = 0x6a09e667, h1 = 0xbb67ae85, h2 = 0x3c6ef372, h3 = 0xa54ff53a;
  let h4 = 0x510e527f, h5 = 0x9b05688c, h6 = 0x1f83d9ab, h7 = 0x5be0cd19;
  const w = new Uint32Array(64);

  for (let off = 0; off < padLen; off += 64) {
    for (let i = 0; i < 16; i++) w[i] = dv.getUint32(off + i * 4, false);
    for (let i = 16; i < 64; i++) {
      const x = w[i - 15], y = w[i - 2];
      const s0 = ((x >>> 7) | (x << 25)) ^ ((x >>> 18) | (x << 14)) ^ (x >>> 3);
      const s1 = ((y >>> 17) | (y << 15)) ^ ((y >>> 19) | (y << 13)) ^ (y >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) >>> 0;
    }
    let a = h0, b = h1, c = h2, d = h3, e = h4, f = h5, g = h6, h = h7;
    for (let i = 0; i < 64; i++) {
      const S1 = ((e >>> 6) | (e << 26)) ^ ((e >>> 11) | (e << 21)) ^ ((e >>> 25) | (e << 7));
      const ch = (e & f) ^ ((~e) & g);
      const t1 = (h + S1 + ch + SHA256_K[i] + w[i]) >>> 0;
      const S0 = ((a >>> 2) | (a << 30)) ^ ((a >>> 13) | (a << 19)) ^ ((a >>> 22) | (a << 10));
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (S0 + maj) >>> 0;
      h = g; g = f; f = e; e = (d + t1) >>> 0;
      d = c; c = b; b = a; a = (t1 + t2) >>> 0;
    }
    h0 = (h0 + a) >>> 0; h1 = (h1 + b) >>> 0; h2 = (h2 + c) >>> 0; h3 = (h3 + d) >>> 0;
    h4 = (h4 + e) >>> 0; h5 = (h5 + f) >>> 0; h6 = (h6 + g) >>> 0; h7 = (h7 + h) >>> 0;
  }

  const out = new Uint8Array(32);
  const outDv = new DataView(out.buffer);
  [h0, h1, h2, h3, h4, h5, h6, h7].forEach((word, idx) => outDv.setUint32(idx * 4, word, false));
  return out;
}

export function sha256Hex(input) {
  const digest = sha256Bytes(input);
  let hex = "";
  for (let i = 0; i < digest.length; i++) hex += digest[i].toString(16).padStart(2, "0");
  return hex;
}

const _f32Buf = new ArrayBuffer(4);
const _f32Dv = new DataView(_f32Buf);

export function toF32(val) {
  _f32Dv.setFloat32(0, Number(val), true);
  return _f32Dv.getFloat32(0, true);
}

export function f32BitsNormalizedZero(val) {
  _f32Dv.setFloat32(0, Number(val), true);
  const f = _f32Dv.getFloat32(0, true);
  if (f === 0) return 0; // Normalize +0.0f and -0.0f to canonical 0x00000000
  return _f32Dv.getUint32(0, true);
}

export function nextafterNegInfF32(val) {
  const f = toF32(val);
  if (!Number.isFinite(f)) return NaN;
  if (f === 0) {
    _f32Dv.setUint32(0, 0x80000001, true); // -Float32.MIN_VALUE
    return _f32Dv.getFloat32(0, true);
  }
  const bits = _f32Dv.getUint32(0, true);
  const nextBits = f > 0 ? (bits - 1) >>> 0 : (bits + 1) >>> 0;
  _f32Dv.setUint32(0, nextBits, true);
  return _f32Dv.getFloat32(0, true);
}

export function encodeFloat16Bits(val) {
  const x = Math.max(-65504, Math.min(65504, Number(val)));
  if (!Number.isFinite(x) || x === 0) return 0;
  const s = x < 0 ? 1 : 0;
  const a = Math.abs(x);
  const e = Math.floor(Math.log2(a));
  if (e < -14) return ((s << 15) | Math.round(a / Math.pow(2, -24))) & 0xffff;
  const expBits = Math.min(30, Math.max(1, e + 15));
  const mant = Math.min(1023, Math.round((a / Math.pow(2, e) - 1) * 1024));
  return ((s << 15) | (expBits << 10) | mant) & 0xffff;
}

export function decodeFloat16Bits(bits) {
  const u = bits & 0xffff;
  const s = (u & 0x8000) ? -1 : 1;
  const e = (u >> 10) & 0x1f;
  const m = u & 0x03ff;
  if (e === 0) return m === 0 ? 0.0 : s * Math.pow(2, -14) * (m / 1024);
  if (e === 31) return m === 0 ? s * Infinity : NaN;
  return s * Math.pow(2, e - 15) * (1 + m / 1024);
}
