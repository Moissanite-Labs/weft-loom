import sys
from math import isqrt

R = (1, 7, 11, 13, 17, 19, 23, 29)


def _small(n):
    if n < 6:
        return (0, 0, 0, 1, 2, 2)[n] if n > 0 else 0
    size = n // 3 + (n % 6 == 2)
    sieve = bytearray([1]) * size
    for i in range(1, isqrt(n) // 3 + 1):
        if sieve[i]:
            k = (3 * i + 1) | 1
            st = 2 * k
            s = k * k // 3
            sieve[s::st] = bytes(len(range(s, size, st)))
            s = k * (k - 2 * (i & 1) + 4) // 3
            sieve[s::st] = bytes(len(range(s, size, st)))
    return sieve.count(1) + 1
